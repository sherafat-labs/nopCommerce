## 2. Initial Nginx Configuration (for Certbot)

First, create a basic HTTP config in `/etc/nginx/sites-available/stoneb2b.ir`:

```nginx
server {
    listen 80;
    server_name stoneb2b.ir www.stoneb2b.ir;
    
    # ACME challenge for Let's Encrypt
    location ^~ /.well-known/acme-challenge/ {
        root /var/www/html;
        default_type "text/plain";
        try_files $uri =404;
    }
    
    # Redirect everything else to HTTPS later
    location / {
        return 301 https://$host$request_uri;
    }
}
```

## 3. Install Certbot and Obtain SSL Certificate

```bash
# Update system
sudo apt update

# Install Certbot
sudo apt install certbot python3-certbot-nginx

# Ensure nginx is running with the basic config
sudo systemctl start nginx
sudo systemctl enable nginx

# Obtain SSL certificate
sudo certbot --nginx -d stoneb2b.ir -d www.stoneb2b.ir
```

During the Certbot process, it will:
- Automatically modify your nginx configuration
- Obtain and install SSL certificates
- Set up automatic redirects from HTTP to HTTPS

## 4. Final Nginx Configuration (Auto-generated by Certbot)

After running Certbot, your `/etc/nginx/sites-available/stoneb2b.ir` will look like:

```nginx
# HTTP redirect to HTTPS
server {
    listen 80;
    server_name stoneb2b.ir www.stoneb2b.ir;
    
    # ACME challenge location for certificate renewal
    location ^~ /.well-known/acme-challenge/ {
        root /var/www/html;
        default_type "text/plain";
        try_files $uri =404;
    }
    
    location / {
        return 301 https://$server_name$request_uri;
    }
}

# HTTPS server
server {
    listen 443 ssl http2;
    server_name stoneb2b.ir www.stoneb2b.ir;

    # SSL certificates by Let's Encrypt
    ssl_certificate /etc/letsencrypt/live/stoneb2b.ir/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/stoneb2b.ir/privkey.pem;
    
    # SSL security settings
    ssl_protocols TLSv1.2 TLSv1.3;
    ssl_ciphers ECDHE-ECDSA-AES128-GCM-SHA256:ECDHE-RSA-AES128-GCM-SHA256:ECDHE-ECDSA-AES256-GCM-SHA384:ECDHE-RSA-AES256-GCM-SHA384;
    ssl_prefer_server_ciphers off;
    ssl_session_cache shared:SSL:10m;
    ssl_session_timeout 10m;

    # Security headers
    add_header X-Frame-Options DENY;
    add_header X-Content-Type-Options nosniff;
    add_header X-XSS-Protection "1; mode=block";
    add_header Strict-Transport-Security "max-age=63072000; includeSubDomains; preload" always;

    # Proxy settings for nopCommerce
    location / {
        proxy_pass http://127.0.0.1:8080;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_set_header X-Forwarded-Host $host;
        proxy_set_header X-Forwarded-Port $server_port;
        
        # Timeout settings
        proxy_connect_timeout 30s;
        proxy_send_timeout 30s;
        proxy_read_timeout 30s;
        
        # Buffer settings
        proxy_buffer_size 128k;
        proxy_buffers 4 256k;
        proxy_busy_buffers_size 256k;
    }

    # Static files caching
    location ~* \.(js|css|png|jpg|jpeg|gif|ico|svg|woff|woff2|ttf|eot)$ {
        proxy_pass http://127.0.0.1:8080;
        proxy_set_header Host $host;
        proxy_hide_header Cache-Control;
        add_header Cache-Control "public, max-age=31536000, immutable";
        
        # Cache static files in nginx
        proxy_cache static_cache;
        proxy_cache_valid 200 302 1d;
        proxy_cache_valid 404 1m;
        proxy_cache_use_stale error timeout updating http_500 http_502 http_503 http_504;
        add_header X-Cache-Status $upstream_cache_status;
    }

    # ACME challenge location for certificate renewal
    location ^~ /.well-known/acme-challenge/ {
        root /var/www/html;
        default_type "text/plain";
        try_files $uri =404;
    }

    # Deny access to . files
    location ~ /\. {
        deny all;
        access_log off;
        log_not_found off;
    }

    access_log /var/log/nginx/stoneb2b.access.log;
    error_log /var/log/nginx/stoneb2b.error.log;
}
```
